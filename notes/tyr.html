SELECT a.*, b.sheet_sta, CONVERT(VARCHAR(10), a.sign_time) AS sign_date, c.reason, CASE WHEN a.sign_time IS NULL THEN '' ELSE CONCAT( DATEDIFF(MINUTE, a.sign_time, ISNULL(a.submit_time, GETDATE())) / 60, '小时', RIGHT('0' + CAST(DATEDIFF(MINUTE, a.sign_time, ISNULL(a.submit_time, GETDATE())) % 60 AS VARCHAR(2)), 2), '分钟' ) END AS last_time, DATEDIFF(DAY, a.sign_time, ISNULL(a.submit_time, GETDATE())) AS last_time2, v.wait_jy_num, d.prdt_no, d.dia, d.ur_diameter, CONCAT('DMP_', pic.file_id) AS small_file_id, dd.dict_name, CASE WHEN dbo.f_get_eligible_for_application_mold_list(d.prdt_no, e.product_line_list, '003', '5') IS NULL THEN '否' ELSE '是' END 当前有模 FROM tb_mold_return_repair_detail a LEFT JOIN tb_mold_return_repair b ON a.sheet_no = b.sheet_no OUTER APPLY ( SELECT STRING_AGG(reason, ',') AS reason FROM tb_mold_reason_list c WHERE c.id IN ( SELECT * FROM STRING_SPLIT(a.return_reason, ',') ) AND type = '返修原因' ) c LEFT JOIN v_mold_wait_jy_num v ON v.mold_no = a.mold_no LEFT JOIN tb_bas_mold d ON a.sn_code = d.sn_code LEFT JOIN zgDmp.dbo.v_dmp_prdt_pic pic ON d.prdt_no = pic.prdt_no LEFT JOIN zgDmp.dbo.tb_bas_dict_detial dd ON d.status = dd.dict_no AND dd.dict_type = '模具库存状态' LEFT JOIN ( SELECT t.prdt_no, STRING_AGG(t.product_line, ',') AS product_line_list FROM ( SELECT DISTINCT b.prdt_no, a.product_line FROM dbo.tb_mes_dispath a WITH (NOLOCK) LEFT JOIN zgErp.dbo.tb_sal_ord2 b WITH (NOLOCK) ON a.ord_lot = b.sheet_lot LEFT JOIN zgDmp.dbo.tb_bas_eqm_line AS TBEL WITH (NOLOCK) ON TBEL.eqm_line_no = a.product_line WHERE a.station_no = 'JY' AND a.work_sta NOT IN ( '2', '99' ) ) t GROUP BY t.prdt_no ) e ON d.prdt_no = e.prdt_no;
